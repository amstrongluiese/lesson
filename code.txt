    <script>
        // --- Configuration & State ---
        const CANVAS_W = 1080;
        const CANVAS_H = 1920;
        
        let bgImage = null;
        let noiseCanvas = null; // Cache for grain

        const state = {
            title: "MIST",
            artist: "ATEEZ",
            timeCurrent: "2:01",
            timeTotal: "3:19",
            blur: 0,
            grain: 40,
            brightness: 85,
            saturation: 80,
            vignette: 60,
            uiOffsetY: 100, // Move default lower to avoid overlapping top pinned components
            uiScale: 1.0,
            showBackButton: true,
            showDetails: true,
            showHeart: true,
            showTimeline: true,
            showControls: true
        };

        // --- Core Elements ---
        const canvas = document.getElementById('renderCanvas');
        const ctx = canvas.getContext('2d');
        canvas.width = CANVAS_W;
        canvas.height = CANVAS_H;

        // Initialize sliders on UI load
        document.getElementById('slUiY').value = state.uiOffsetY;
        document.getElementById('valUiY').textContent = state.uiOffsetY;

        // Ensure fonts are loaded before first render
        document.fonts.ready.then(() => {
            render();
        });

        // --- Event Handlers ---
        function updateState(key, value) {
            state[key] = value;
            requestAnimationFrame(render);
        }

        function handleImageUpload(event) {
            const file = event.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(e) {
                const img = new Image();
                img.onload = function() {
                    bgImage = img;
                    requestAnimationFrame(render);
                }
                img.src = e.target.result;
            }
            reader.readAsDataURL(file);
        }

        // --- Helper: Generate Grain ---
        function generateNoise() {
            if (noiseCanvas) return noiseCanvas;
            noiseCanvas = document.createElement('canvas');
            noiseCanvas.width = CANVAS_W / 2; // Generate at half res for speed
            noiseCanvas.height = CANVAS_H / 2;
            const nCtx = noiseCanvas.getContext('2d');
            const imgData = nCtx.createImageData(noiseCanvas.width, noiseCanvas.height);
            const data = imgData.data;
            for (let i = 0; i < data.length; i += 4) {
                const val = Math.random() * 255;
                data[i] = val;     // R
                data[i+1] = val;   // G
                data[i+2] = val;   // B
                data[i+3] = 255;   // A
            }
            nCtx.putImageData(imgData, 0, 0);
            return noiseCanvas;
        }

        // --- Helper: Convert Time string "M:SS" to seconds ---
        function timeToSeconds(timeStr) {
            const parts = timeStr.split(':');
            if (parts.length !== 2) return 0;
            return parseInt(parts[0]) * 60 + parseInt(parts[1]) || 0;
        }

        // --- SVG Path Drawer Helper ---
        function drawPath(ctx, pathStr, x, y, scale, strokeColor, fillColor, strokeWidth = 2, lineCap = 'round', lineJoin = 'round') {
            ctx.save();
            ctx.translate(x, y);
            ctx.scale(scale, scale);
            const p = new Path2D(pathStr);
            if (fillColor) {
                ctx.fillStyle = fillColor;
                ctx.fill(p);
            }
            if (strokeColor) {
                ctx.strokeStyle = strokeColor;
                ctx.lineWidth = strokeWidth / scale; // Keep absolute stroke width
                ctx.lineCap = lineCap;
                ctx.lineJoin = lineJoin;
                ctx.stroke(p);
            }
            ctx.restore();
        }

        // --- Main Render Engine ---
        function render() {
            // 1. Clear Canvas
            ctx.clearRect(0, 0, CANVAS_W, CANVAS_H);

            // 2. Draw Background
            if (bgImage) {
                // Apply CSS Filters
                ctx.filter = `blur(${state.blur}px) brightness(${state.brightness}%) saturate(${state.saturation}%)`;
                
                // Calculate cover aspect ratio
                const scale = Math.max(CANVAS_W / bgImage.width, CANVAS_H / bgImage.height);
                const x = (CANVAS_W / 2) - (bgImage.width / 2) * scale;
                const y = (CANVAS_H / 2) - (bgImage.height / 2) * scale;
                
                ctx.drawImage(bgImage, x, y, bgImage.width * scale, bgImage.height * scale);
                ctx.filter = 'none'; // reset
            } else {
                // Premium default placeholder gradient
                const grad = ctx.createLinearGradient(0,0, 0, CANVAS_H);
                grad.addColorStop(0, '#1c1b26');
                grad.addColorStop(0.5, '#0e0e13');
                grad.addColorStop(1, '#050508');
                ctx.fillStyle = grad;
                ctx.fillRect(0,0, CANVAS_W, CANVAS_H);
            }

            // 3. Apply Effects (Grain & Dark Gradient/Vignette)
            if (state.grain > 0) {
                const noise = generateNoise();
                ctx.globalAlpha = state.grain / 300; // Adjust opacity intensity
                ctx.globalCompositeOperation = 'overlay';
                ctx.drawImage(noise, 0, 0, CANVAS_W, CANVAS_H);
                ctx.globalCompositeOperation = 'source-over';
                ctx.globalAlpha = 1.0;
            }

            if (state.vignette > 0) {
                // Aesthetic soft dark glow around screen borders
                const radialGrad = ctx.createRadialGradient(
                    CANVAS_W/2, CANVAS_H/2, CANVAS_W * 0.4, 
                    CANVAS_W/2, CANVAS_H/2, CANVAS_H * 0.75
                );
                radialGrad.addColorStop(0, 'rgba(0,0,0,0)');
                radialGrad.addColorStop(1, `rgba(0,0,0,${state.vignette / 100 * 0.85})`);
                ctx.fillStyle = radialGrad;
                ctx.fillRect(0, 0, CANVAS_W, CANVAS_H);

                // Deep dark linear gradient anchored at the bottom to hold text cleanly
                const linearGrad = ctx.createLinearGradient(0, CANVAS_H * 0.35, 0, CANVAS_H);
                linearGrad.addColorStop(0, 'rgba(0,0,0,0)');
                linearGrad.addColorStop(1, `rgba(0,0,0,${state.vignette / 100 * 0.95})`);
                ctx.fillStyle = linearGrad;
                ctx.fillRect(0, 0, CANVAS_W, CANVAS_H);
            }

            // ==========================================
            // 4. DRAW Pinned Static Back Button
            // ==========================================
            const marginX = 90; 
            
            if (state.showBackButton) {
                const backBtnSize = 110;
                const backBtnY = 120;
                
                // Draw translucent glass rounded capsule
                ctx.fillStyle = "rgba(18, 18, 24, 0.45)";
                ctx.strokeStyle = "rgba(255, 255, 255, 0.08)";
                ctx.lineWidth = 3;
                ctx.beginPath();
                ctx.roundRect(marginX, backBtnY, backBtnSize, backBtnSize, 36);
                ctx.fill();
                ctx.stroke();
                
                // Draw standard iOS styled '<' icon inside back button
                ctx.strokeStyle = "#ffffff";
                ctx.lineWidth = 6;
                ctx.lineCap = "round";
                ctx.lineJoin = "round";
                ctx.beginPath();
                ctx.moveTo(marginX + 60, backBtnY + 34);
                ctx.lineTo(marginX + 40, backBtnY + 55);
                ctx.lineTo(marginX + 60, backBtnY + 76);
                ctx.stroke();
            }

            // ==========================================
            // 5. DRAW INTERACTIVE MUSIC PLAYER CONTROL PANEL
            // ==========================================
            ctx.save();
            
            // Apply UI Transformations (Move & Scale relative to center)
            ctx.translate(CANVAS_W / 2, CANVAS_H / 2 + state.uiOffsetY);
            ctx.scale(state.uiScale, state.uiScale);
            ctx.translate(-CANVAS_W / 2, -CANVAS_H / 2);

            const textY = 1000;

            // --- Track Title & Artist ---
            if (state.showDetails) {
                // Song Title (Extra Bold Minimalist Sans)
                ctx.fillStyle = "#ffffff";
                ctx.font = "900 68px 'Inter', sans-serif";
                ctx.textAlign = "left";
                ctx.fillText(state.title.toUpperCase(), marginX, textY);
                
                // Artist Name
                ctx.font = "600 42px 'Inter', sans-serif";
                ctx.fillStyle = "rgba(255, 255, 255, 0.75)";
                ctx.fillText(state.artist.toUpperCase(), marginX, textY + 70);
            }

            // --- Elegant Outline Heart Icon (Right-aligned next to Title) ---
            if (state.showHeart) {
                const heartPath = "M12 21.35l-1.45-1.32C5.4 15.36 2 12.28 2 8.5 2 5.42 4.42 3 7.5 3c1.74 0 3.41.81 4.5 2.09C13.09 3.81 14.76 3 16.5 3 19.58 3 22 5.42 22 8.5c0 3.78-3.4 6.86-8.55 11.54L12 21.35z";
                drawPath(ctx, heartPath, CANVAS_W - marginX - 55, textY + 5, 2.5, null, "#ffffff");
            }

            // --- Progress Timeline ---
            const barY = 1190;
            const barWidth = CANVAS_W - (marginX * 2);

            if (state.showTimeline) {
                // Calculate progress percentage
                const curSec = timeToSeconds(state.timeCurrent);
                const totSec = timeToSeconds(state.timeTotal);
                let progress = totSec > 0 ? curSec / totSec : 0;
                if (progress > 1) progress = 1;
                if (progress < 0) progress = 0;

                const activeWidth = barWidth * progress;

                // Translucent Timeline Track Line
                ctx.fillStyle = "rgba(255, 255, 255, 0.16)";
                ctx.beginPath();
                ctx.roundRect(marginX, barY, barWidth, 8, 4);
                ctx.fill();

                // Highlighted Active Progress Track Line
                ctx.fillStyle = "#ffffff";
                ctx.beginPath();
                ctx.roundRect(marginX, barY, activeWidth, 8, 4);
                ctx.fill();

                // Progress Slider Knob (Clean white circle dot)
                ctx.beginPath();
                ctx.arc(marginX + activeWidth, barY + 4, 15, 0, Math.PI * 2);
                ctx.fill();

                // --- Timestamps ---
                ctx.font = "500 28px 'Inter', sans-serif";
                ctx.fillStyle = "rgba(255, 255, 255, 0.6)";
                ctx.textAlign = "left";
                ctx.fillText(state.timeCurrent, marginX, barY + 52);
                
                ctx.textAlign = "right";
                ctx.fillText(state.timeTotal, CANVAS_W - marginX, barY + 52);
            }

            // --- Premium Media Playback Controls ---
            if (state.showControls) {
                const controlsY = 1360;
                const centerY = controlsY + 45;

                // 1. Shuffle Switch
                const shufflePath = "M16 3h5v5M4 20L21 3M21 16v5h-5M15 15l6 6M4 4l5 5";
                drawPath(ctx, shufflePath, marginX, centerY - 25, 2.3, "rgba(255,255,255,0.75)", null, 2.5);

                // 2. Skip Backward
                const prevPath = "M19 20L9 12l10-8v16zM5 19V5h2v14H5z";
                drawPath(ctx, prevPath, 340, centerY - 28, 2.5, null, "#ffffff");

                // 3. Iconic Center Play Button Ring
                const playCircleX = CANVAS_W / 2;
                
                // Translucent glass base behind the play control
                ctx.fillStyle = "rgba(255, 255, 255, 0.08)";
                ctx.beginPath();
                ctx.arc(playCircleX, centerY, 90, 0, Math.PI * 2);
                ctx.fill();

                // Main White Circle Button
                ctx.fillStyle = "#ffffff";
                ctx.beginPath();
                ctx.arc(playCircleX, centerY, 75, 0, Math.PI * 2);
                ctx.fill();
                
                // Pause columns inside button
                ctx.fillStyle = "#121216";
                ctx.beginPath();
                ctx.roundRect(playCircleX - 24, centerY - 25, 16, 50, 4);
                ctx.fill();
                ctx.beginPath();
                ctx.roundRect(playCircleX + 8, centerY - 25, 16, 50, 4);
                ctx.fill();

                // 4. Skip Forward
                const nextPath = "M5 4l10 8-10 8V4zM19 5v14h-2V5h2z";
                drawPath(ctx, nextPath, CANVAS_W - 340 - 60, centerY - 28, 2.5, null, "#ffffff");

                // 5. Repeat Track Switch
                const repeatPath = "M17 2l4 4-4 4M3 11v-1a4 4 0 0 1 4-4h14M7 22l-4-4 4-4M21 13v1a4 4 0 0 1-4 4H3";
                drawPath(ctx, repeatPath, CANVAS_W - marginX - 55, centerY - 25, 2.3, "rgba(255,255,255,0.75)", null, 2.5);
            }
            
            ctx.restore();
        }

        // --- Interactive Gesture Controls on Canvas (Drag & Scroll) ---
        let isDragging = false;
        let startY = 0;
        let initialOffsetY = 0;

        canvas.style.cursor = 'grab';

        function handleDragStart(clientY) {
            isDragging = true;
            startY = clientY;
            initialOffsetY = state.uiOffsetY;
            canvas.style.cursor = 'grabbing';
        }

        function handleDragMove(clientY) {
            if (!isDragging) return;
            const rect = canvas.getBoundingClientRect();
            
            // Adjust calculation based on standard scale of the canvas preview
            const displayScale = CANVAS_H / rect.height;
            const deltaY = (clientY - startY) * displayScale;
            
            // Boundaries check so player doesn't fly off screen completely
            const newY = Math.max(-950, Math.min(950, initialOffsetY + deltaY));
            updateState('uiOffsetY', newY);
            
            // Update input range slider inside control side-panel
            const slider = document.getElementById('slUiY');
            if (slider) {
                slider.value = newY;
                document.getElementById('valUiY').textContent = Math.round(newY);
            }
        }

        function handleDragEnd() {
            isDragging = false;
            canvas.style.cursor = 'grab';
        }

        // Attach Mouse Drag Observers
        canvas.addEventListener('mousedown', (e) => handleDragStart(e.clientY));
        window.addEventListener('mousemove', (e) => handleDragMove(e.clientY));
        window.addEventListener('mouseup', handleDragEnd);

        // Attach Touch Screen Swipe Observers
        canvas.addEventListener('touchstart', (e) => handleDragStart(e.touches[0].clientY), {passive: true});
        window.addEventListener('touchmove', (e) => {
            if (isDragging) {
                if (e.cancelable) e.preventDefault(); 
                handleDragMove(e.touches[0].clientY);
            }
        }, {passive: false});
        window.addEventListener('touchend', handleDragEnd);

        // Scroll Wheel Mouse Zoom/Scale Adjuster
        canvas.addEventListener('wheel', (e) => {
            e.preventDefault();
            const scaleChange = e.deltaY > 0 ? -0.04 : 0.04;
            let newScale = Math.max(0.4, Math.min(1.6, state.uiScale + scaleChange));
            
            newScale = Math.round(newScale * 100) / 100;
            updateState('uiScale', newScale);
            
            const slider = document.getElementById('slUiScale');
            if (slider) {
                slider.value = newScale;
                document.getElementById('valUiScale').textContent = newScale.toFixed(2) + 'x';
            }
        }, {passive: false});
    </script>
