    <script>
        document.addEventListener('DOMContentLoaded', () => {
            
            // --- 1. Cinematic Loader ---
            const loader = document.getElementById('cinematic-loader');
            const text1 = document.getElementById('loader-text-1');
            const text2 = document.getElementById('loader-text-2');
            
            setTimeout(() => { text1.style.opacity = '1'; }, 500);
            setTimeout(() => { text2.style.opacity = '1'; }, 2000);
            setTimeout(() => {
                loader.style.opacity = '0';
                setTimeout(() => { loader.style.display = 'none'; }, 1500);
            }, 4500);

            // --- 2. Custom Cursor ---
            const cursorDot = document.getElementById('cursor-dot');
            const cursorOutline = document.getElementById('cursor-outline');
            let mouseX = window.innerWidth / 2;
            let mouseY = window.innerHeight / 2;
            let outlineX = mouseX;
            let outlineY = mouseY;

            // Hide standard cursor on non-touch devices
            if (window.matchMedia("(pointer: fine)").matches) {
                window.addEventListener('mousemove', (e) => {
                    mouseX = e.clientX;
                    mouseY = e.clientY;
                    cursorDot.style.left = `${mouseX}px`;
                    cursorDot.style.top = `${mouseY}px`;
                });

                // Smooth following outline
                const animateCursor = () => {
                    let dx = mouseX - outlineX;
                    let dy = mouseY - outlineY;
                    outlineX += dx * 0.15;
                    outlineY += dy * 0.15;
                    cursorOutline.style.left = `${outlineX}px`;
                    cursorOutline.style.top = `${outlineY}px`;
                    requestAnimationFrame(animateCursor);
                };
                animateCursor();

                // Hover effects for cursor
                const interactables = document.querySelectorAll('a, button, .polaroid, .glass-card');
                interactables.forEach(el => {
                    el.addEventListener('mouseenter', () => {
                        cursorOutline.style.transform = 'translate(-50%, -50%) scale(1.5)';
                        cursorOutline.style.backgroundColor = 'rgba(84, 92, 82, 0.1)'; // moss tint
                        cursorOutline.style.borderColor = 'rgba(84, 92, 82, 0.5)';
                    });
                    el.addEventListener('mouseleave', () => {
                        cursorOutline.style.transform = 'translate(-50%, -50%) scale(1)';
                        cursorOutline.style.backgroundColor = 'transparent';
                        cursorOutline.style.borderColor = 'rgba(44, 46, 44, 0.4)';
                    });
                });
            } else {
                cursorDot.style.display = 'none';
                cursorOutline.style.display = 'none';
                document.body.style.cursor = 'auto';
            }

            // --- 3. Scroll Reveal & Progress Bar ---
            const reveals = document.querySelectorAll('.reveal');
            const progressBar = document.getElementById('scroll-progress');

            const revealOnScroll = () => {
                let winScroll = document.body.scrollTop || document.documentElement.scrollTop;
                let height = document.documentElement.scrollHeight - document.documentElement.clientHeight;
                let scrolled = (winScroll / height) * 100;
                progressBar.style.width = scrolled + "%";

                const windowHeight = window.innerHeight;
                reveals.forEach(reveal => {
                    const revealTop = reveal.getBoundingClientRect().top;
                    const revealPoint = 100; // triggers 100px before appearing
                    if (revealTop < windowHeight - revealPoint) {
                        reveal.classList.add('active');
                    }
                });
            };
            window.addEventListener('scroll', revealOnScroll);
            revealOnScroll(); // Trigger once on load

            // --- 4. Interactive Particles (Fireflies/Dust) ---
            const canvas = document.getElementById('particle-canvas');
            const ctx = canvas.getContext('2d');
            let particlesArray = [];
            
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;

            window.addEventListener('resize', () => {
                canvas.width = window.innerWidth;
                canvas.height = window.innerHeight;
                initParticles();
            });

            class Particle {
                constructor() {
                    this.x = Math.random() * canvas.width;
                    this.y = Math.random() * canvas.height;
                    this.size = Math.random() * 2 + 0.5;
                    this.speedX = Math.random() * 1 - 0.5;
                    this.speedY = Math.random() * 1 - 0.5;
                    // Colors: soft whites, pale golds
                    this.color = Math.random() > 0.5 ? 'rgba(255, 255, 255, 0.6)' : 'rgba(209, 201, 184, 0.6)';
                }
                update() {
                    this.x += this.speedX;
                    this.y += this.speedY;

                    // Slight interaction with mouse if on desktop
                    if (window.matchMedia("(pointer: fine)").matches) {
                        let dx = mouseX - this.x;
                        let dy = mouseY - this.y;
                        let distance = Math.sqrt(dx * dx + dy * dy);
                        if (distance < 100) {
                            // gently push away
                            this.x -= dx * 0.01;
                            this.y -= dy * 0.01;
                        }
                    }

                    if (this.x < 0 || this.x > canvas.width) this.speedX *= -1;
                    if (this.y < 0 || this.y > canvas.height) this.speedY *= -1;
                }
                draw() {
                    ctx.fillStyle = this.color;
                    ctx.beginPath();
                    ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                    ctx.fill();
                    
                    // Add subtle glow
                    ctx.shadowBlur = 8;
                    ctx.shadowColor = this.color;
                }
            }

            function initParticles() {
                particlesArray = [];
                let numberOfParticles = (canvas.width * canvas.height) / 15000; // Density
                for (let i = 0; i < numberOfParticles; i++) {
                    particlesArray.push(new Particle());
                }
            }
            
            function animateParticles() {
                ctx.clearRect(0, 0, canvas.width, canvas.height);
                for (let i = 0; i < particlesArray.length; i++) {
                    particlesArray[i].update();
                    particlesArray[i].draw();
                }
                requestAnimationFrame(animateParticles);
            }
            
            initParticles();
            animateParticles();

            // --- 5. Audio Handling (Background & Ambient) ---
            const bgMusic = document.getElementById('bg-music');
            const ambientMusic = document.getElementById('ambient-music');
            const musicToggleBtn = document.getElementById('music-toggle');
            let isMusicPlaying = false;

            // Set volumes
            bgMusic.volume = 0.4;
            ambientMusic.volume = 0.2;

            function playAudio() {
                // Browser policies require user interaction before playing audio.
                // We catch the promises to prevent Unhandled Promise Rejection errors.
                let bgPromise = bgMusic.play();
                let ambPromise = ambientMusic.play();

                if (bgPromise !== undefined) {
                    bgPromise.then(_ => {
                        isMusicPlaying = true;
                        updateAudioIcon();
                    }).catch(error => {
                        console.log("Autoplay prevented by browser. Waiting for explicit click.");
                        isMusicPlaying = false;
                        updateAudioIcon();
                    });
                }

                if (ambPromise !== undefined) {
                    ambPromise.catch(error => {
                        console.log("Ambient autoplay prevented by browser.");
                    });
                }
            }

            function toggleAudio() {
                if (isMusicPlaying) {
                    bgMusic.pause();
                    ambientMusic.pause();
                    isMusicPlaying = false;
                    updateAudioIcon();
                } else {
                    playAudio();
                }
            }

            function updateAudioIcon() {
                if (isMusicPlaying) {
                    // Sound Waves icon
                    musicToggleBtn.innerHTML = `
                        <svg class="w-6 h-6 text-moss animate-pulse" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                            <path stroke-linecap="round" stroke-linejoin="round" stroke-width="1.5" d="M9 19V6l12-3v13M9 19c0 1.105-1.343 2-3 2s-3-.895-3-2 1.343-2 3-2 3 .895 3 2zm12-3c0 1.105-1.343 2-3 2s-3-.895-3-2 1.343-2 3-2 3 .895 3 2zM9 10l12-3"></path>
                        </svg>`;
                } else {
                    // Muted icon
                    musicToggleBtn.innerHTML = `
                        <svg class="w-6 h-6 text-charcoal opacity-70" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                            <path stroke-linecap="round" stroke-linejoin="round" stroke-width="1.5" d="M5.586 15H4a1 1 0 01-1-1v-4a1 1 0 011-1h1.586l4.707-4.707C10.923 3.663 12 4.109 12 5v14c0 .891-1.077 1.337-1.707.693L5.586 15z" clip-rule="evenodd" />
                            <path stroke-linecap="round" stroke-linejoin="round" stroke-width="1.5" d="M17 14l2-2m0 0l2-2m-2 2l-2-2m2 2l2 2"/>
                        </svg>`;
                }
            }

            // Attempt autoplay on first meaningful interaction (scroll, click, touch)
            window.addEventListener('scroll', playAudio, { once: true });
            document.body.addEventListener('click', playAudio, { once: true });
            document.body.addEventListener('touchstart', playAudio, { once: true });
            
            // Manual toggle button
            musicToggleBtn.addEventListener('click', (e) => {
                e.stopPropagation(); // prevent triggering the body click above
                toggleAudio();
            });

            // --- 6. Parallax effect for Collage (Subtle) ---
            const collage = document.getElementById('collage');
            window.addEventListener('mousemove', (e) => {
                if(window.matchMedia("(pointer: fine)").matches) {
                    const x = (e.clientX / window.innerWidth - 0.5) * 20;
                    const y = (e.clientY / window.innerHeight - 0.5) * 20;
                    collage.style.transform = `translate(${x}px, ${y}px)`;
                }
            });
        });
